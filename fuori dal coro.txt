const UNMATCHED_PCT = 1;

/* NORMALIZZAZIONE E MATCHING */
function norm(s){
  s = s.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g,"");
  s = s.replace(/['’`]/g," ").replace(/[^a-z0-9\s]/g," ").replace(/\s+/g," ").trim();
  for (let k=0;k<2;k++) s = s.replace(/^(il|lo|la|le|i|gli|un|una|uno|l)\s+/,"");
  return s.trim();
}

function match(raw, q) {
  const clean = norm(raw);
  if(!clean) return null;
  for (let item of q.items) {
    if(item.alias.some(a => norm(a) === clean)) return item;
  }
  return null;
}

function calcPoints(mode, pct, maxScore) {
  if (mode === "nel") {
    return Math.round(100 * (pct / maxScore));
  } else {
    return Math.round(100 * (1 - (pct / maxScore)));
  }
}

const KEY = "fdc_daily_v1";
let st = { streak: 0, last: null, answers: [] };
try { const s = JSON.parse(localStorage.getItem(KEY)); if(s) st = s; } catch(e){}
const persist = () => { try { localStorage.setItem(KEY, JSON.stringify(st)); } catch(e){} };

function dayKey(off=0){
  const d=new Date(); d.setDate(d.getDate()+off);
  return d.getFullYear()+"-"+String(d.getMonth()+1).padStart(2,"0")+"-"+String(d.getDate()).padStart(2,"0");
}
const streakNow = () => (st.last===dayKey() || st.last===dayKey(-1)) ? st.streak : 0;
function drawStreak(){ document.getElementById("streak").innerHTML = "Serie: <b>"+streakNow()+"</b> "+(streakNow()===1?"giorno":"giorni"); }

  fuori: "Obiettivo: trova le risposte più RARE. Meno persone la pensano come te, più punti fai!",
  nel: "Obiettivo: trova le risposte più COMUNI. Più persone la pensano come te, più punti fai!"
};

function setMode(m){
  currentMode = m; 
  document.documentElement.dataset.mode = m;
  $("m-fuori").setAttribute("aria-pressed", m === "fuori");
  $("m-nel").setAttribute("aria-pressed", m === "nel");
  $("hint").textContent = HINT[m];
}

function renderProgress(){
  const box = $("progress");
  box.innerHTML = "";
  QUESTIONS.forEach((_, i) => {
    const d = document.createElement("div");
    d.className = "progress-dot" + (i <= currentStep ? " active" : "");
    box.appendChild(d);
  });
}

function showPlay(){
  $("play").hidden = false; 
  $("res").hidden = true;
  drawStreak();
  renderProgress();
  $("q").textContent = QUESTIONS[currentStep].text;
  $("ans").value = "";
  setMode(currentMode);
}

function submitAnswer(){
  const raw = $("ans").value.trim();
  if(raw.length < 2){ 
    $("err").textContent = "Inserisci una risposta valida."; 
    return; 
  }
  $("err").textContent = "";

  const q = QUESTIONS[currentStep];
  const matched = match(raw, q);
  const pct = matched ? matched.score : UNMATCHED_PCT;
  const pts = calcPoints(currentMode, pct, q.maxScore);

  sessionAnswers.push({
    qText: q.text,
    raw: raw,
    canon: matched ? matched.canonical : raw,
    pct: pct,
    pts: pts
  });

  if (currentStep + 1 < QUESTIONS.length) {
    currentStep++;
    showPlay();
  } else {
    finishGame();
  }
}

function finishGame(){
  const t = dayKey();
  if(st.last !== t){
    st.streak = (st.last === dayKey(-1)) ? st.streak + 1 : 1;
    st.last = t;
  }
  st.answers = sessionAnswers;
  st.mode = currentMode;
  persist();
  showResults();
}

function showResults(){
  $("play").hidden = true; 
  $("res").hidden = false;
  drawStreak();

  const totalPts = sessionAnswers.reduce((acc, a) => acc + a.pts, 0);
  $("pts").textContent = totalPts;
  $("modeSummary").textContent = "Modalità giocata: " + (currentMode === "nel" ? "Nel Coro" : "Fuori dal Coro");

  if(currentMode === "fuori") {
    $("msg").textContent = totalPts > 350 ? "Sei un vero spirito libero! 🦄" : "Ti muovi un po' troppo con la massa...";
  } else {
    $("msg").textContent = totalPts > 350 ? "In piena sintonia con la massa! 🐑" : "Troppo originale per stare nel coro!";
  }

  const list = $("summaryList");
  list.innerHTML = "";
  sessionAnswers.forEach(a => {
    const li = document.createElement("li");
    li.className = "summary-item";
    li.innerHTML = `
      <div>
        <strong>${a.canon}</strong>
        <span>Scelta dal ${a.pct}% degli italiani</span>
      </div>
      <div class="summary-score">+${a.pts}</div>
    `;
    list.appendChild(li);
  });
}

async function share(){

  const d = new Date().toLocaleDateString("it-IT", {day:"numeric", month:"short"});

  let grid = sessionAnswers.map(a => {
    if(a.pts >= 80) return "🟩";
    if(a.pts >= 40) return "🟨";
    return "🟥";
  }).join("");

  const shareText = `Fuori dal Coro / Nel Coro (${d})\nModalità: ${modeLabel}\nPunteggio: ${totalPts}/500 pt\n${grid}\n\nGioca anche tu!`;

  try {
    if(navigator.share){ 
      await navigator.share({ text: shareText });

    await navigator.clipboard.writeText(shareText);

  } catch(e) {
    $("shareNote").textContent = "Copia questo testo per condividere:\n" + shareText;
  }
}
